<script>
//scoreboard
let board = Array(9).fill('');
let moveOrder = [];
let currentPlayer = 'X';
let gameOver = false;
let score = {X:0,O:0};
let scoreOn = false;
//all possible ways to win
const wins=[
 [0,1,2],[3,4,5],[6,7,8],
 [0,3,6],[1,4,7],[2,5,8],
 [0,4,8],[2,4,6]
];

const boardEl=document.getElementById('board');
const statusEl=document.getElementById('status');
const scoreEl=document.getElementById('score');

function render(){
  boardEl.innerHTML='';
  board.forEach((v,i)=>{
    const c=document.createElement('div');
    c.className='cell';
    if(v.startsWith('X'))c.classList.add('x');
    if(v.startsWith('O'))c.classList.add('o');
    if(v.endsWith('-fade'))c.classList.add('fade');
    c.textContent=v.replace('-fade','');
    c.onclick=()=>move(i);
    boardEl.appendChild(c);
  });
}

async function move(i){
  if(gameOver) return;
  if(board[i]) return;

  // for fading oldest move
  if(board.filter(v=>v).length === 7){
    const old = moveOrder.shift();
    board[old] += '-fade';
    render();

    await new Promise(r=>setTimeout(r,450));
    if(gameOver) return;

    board[old]='';
    render();
  }

  board[i]=currentPlayer;
  moveOrder.push(i);
  render();

  const win=checkWin();
  if(win){
    gameOver=true;
    statusEl.textContent=`Player ${currentPlayer} wins!`;
    win.forEach(i=>boardEl.children[i].classList.add('win'));
    score[currentPlayer]++;
    updateScore();
    return;
  }

  currentPlayer=currentPlayer==='X'?'O':'X';
  statusEl.textContent=`Player ${currentPlayer}'s turn`;
}

function checkWin(){
  for(const [a,b,c] of wins){
    const A=board[a]?.replace('-fade','');
    const B=board[b]?.replace('-fade','');
    const C=board[c]?.replace('-fade','');
    if(A&&A===B&&A===C) return[a,b,c];
  }
  return null;
}

function updateScore(){
  scoreEl.textContent=`Score — X: ${score.X} | O: ${score.O}`;
}

document.getElementById('reset').onclick=()=>{
  board=Array(9).fill('');
  moveOrder=[];
  currentPlayer='X';
  gameOver=false;
  statusEl.textContent="Player X's turn";
  render();
};

document.getElementById('toggleScore').onclick=()=>{
  scoreOn=!scoreOn;
  scoreEl.style.display=scoreOn?'block':'none';
  if(scoreOn) updateScore();
};

render();
</script>
